cmake_minimum_required(VERSION 3.10)
project(CalculatriceDynamique C)

# Configuration
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -g")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# Répertoires personnalisés
set(DYNAMIC_BUILD_DIR ${CMAKE_BINARY_DIR}/buildForDynamiqueLib)
set(BIN_DIR ${CMAKE_BINARY_DIR}/bin)

# Configuration des outputs par cible
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${DYNAMIC_BUILD_DIR})
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${DYNAMIC_BUILD_DIR})
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${BIN_DIR})

# Fonction pour créer une librairie dynamique avec ses includes
function(add_dynamic_library lib_name source_file)
    add_library(${lib_name} SHARED ${source_file})

    # Définir le répertoire de sortie spécifique
    set_target_properties(${lib_name} PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY ${DYNAMIC_BUILD_DIR}/${lib_name}
        ARCHIVE_OUTPUT_DIRECTORY ${DYNAMIC_BUILD_DIR}/${lib_name}
    )

    # Includes publics
    target_include_directories(${lib_name} PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/src/lib/calstatlib/${lib_name}
        ${CMAKE_CURRENT_SOURCE_DIR}/src/lib/calstatlib/tokenType
    )

    # Gestion des dépendances mathématiques
    if(${lib_name} STREQUAL "evalRpn" OR ${lib_name} STREQUAL "tokenize")
        target_link_libraries(${lib_name} m)
    endif()
endfunction()

# Création des bibliothèques dynamiques
add_dynamic_library(evalRpn src/lib/calstatlib/evalRpn/evalRpn.c)
add_dynamic_library(fixUnaryMinus src/lib/calstatlib/fixUnaryMinus/fixUnaryMinus.c)
add_dynamic_library(shuntingYard src/lib/calstatlib/shuntingYard/shuntingYard.c)
add_dynamic_library(tokenize src/lib/calstatlib/tokenize/tokenize.c)

# Exécutable
add_executable(exeWithDynLib src/app/main.c)

# Configuration des includes pour l'exécutable
target_include_directories(exeWithDynLib PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src/lib/calstatlib/tokenize
    ${CMAKE_CURRENT_SOURCE_DIR}/src/lib/calstatlib/shuntingYard
    ${CMAKE_CURRENT_SOURCE_DIR}/src/lib/calstatlib/fixUnaryMinus
    ${CMAKE_CURRENT_SOURCE_DIR}/src/lib/calstatlib/evalRpn
    ${CMAKE_CURRENT_SOURCE_DIR}/src/lib/calstatlib/tokenType
    # Inclure les répertoires de build pour les headers générés si nécessaire
)

# Liaison des bibliothèques
target_link_libraries(exeWithDynLib
    evalRpn
    fixUnaryMinus
    shuntingYard
    tokenize
    m
)

# Configuration RPATH pour le développement
if(UNIX AND NOT APPLE)
    # Utiliser RPATH pour trouver les libs dans le build directory
    set_target_properties(exeWithDynLib PROPERTIES
        INSTALL_RPATH "$ORIGIN/../buildForDynamiqueLib"
        BUILD_WITH_INSTALL_RPATH FALSE
        SKIP_BUILD_RPATH FALSE
    )

    # Ajouter les chemins des bibliothèques au RPATH
    foreach(lib evalRpn fixUnaryMinus shuntingYard tokenize)
        set_target_properties(exeWithDynLib PROPERTIES
            BUILD_RPATH "${DYNAMIC_BUILD_DIR}/${lib};${BUILD_RPATH}"
        )
    endforeach()
endif()

# Cible pour exécuter le programme
add_custom_target(run
    COMMAND ./exeWithDynLib
    WORKING_DIRECTORY ${BIN_DIR}
    DEPENDS exeWithDynLib
    COMMENT "Exécution du programme avec bibliothèques dynamiques"
)
